Incantation

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

#define PORT 8080

int main(void) {
    /* Create a TCP socket (IPv4, stream). */
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd < 0) {
        perror("socket");
        return EXIT_FAILURE;
    }

    /* Allow quick restarts without "Address already in use". */
    int opt = 1;
    setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof opt);

    /* Bind to every local interface on PORT. */
    struct sockaddr_in addr;
    memset(&addr, 0, sizeof addr);
    addr.sin_family = AF_INET;
    addr.sin_addr.s_addr = htonl(INADDR_ANY);
    addr.sin_port = htons(PORT);

    if (bind(server_fd, (struct sockaddr *)&addr, sizeof addr) < 0) {
        perror("bind");
        close(server_fd);
        return EXIT_FAILURE;
    }

    if (listen(server_fd, 1) < 0) {
        perror("listen");
        close(server_fd);
        return EXIT_FAILURE;
    }

    printf("Circle drawn. Listening on port %d...\n", PORT);
    fflush(stdout);

    /* Block until one client connects, greet it, then close. */
    int client_fd = accept(server_fd, NULL, NULL);
    if (client_fd < 0) {
        perror("accept");
        close(server_fd);
        return EXIT_FAILURE;
    }

    const char *msg = "The ritual is complete.\n";
    send(client_fd, msg, strlen(msg), 0);

    close(client_fd);
    close(server_fd);
    return EXIT_SUCCESS;
}

Effect

In plain terms: Starts a TCP server on port 8080, waits for one client to connect, sends it one line of text, and exits.

The forge-mage draws a circle on port 8080 and waits in silence. The first soul to step inside receives a single message, and the circle closes behind them.

socket() creates an IPv4 TCP endpoint. SO_REUSEADDR lets you restart the program right away without the "Address already in use" backfire. bind() attaches the socket to every local interface (INADDR_ANY) on port 8080, using htons() and htonl() to convert numbers to network byte order. listen() marks it as a server, accept() blocks until a client arrives, and send() writes the message before both sockets are closed.

Side Effects

  • Opens port 8080 on all network interfaces while it runs. Other machines on your network can connect. Change INADDR_ANY to htonl(INADDR_LOOPBACK) to allow local connections only.
  • Backfires with bind: Address already in use if another program already holds port 8080.
  • Serves exactly one client, then exits. Wrap accept() in a loop for a server that keeps running.
